# ==========================================
# 1. 策略配置 (JSON) - 均线突破策略 V4
# ==========================================
STRATEGY_JSON = """
{

    },

        "max_spread_pct": 0.005,    
        
        "macd_fast": 12,           
        "macd_slow": 26,
        "macd_signal": 9,
        "rsi_period": 14,
        "rsi_profit_exit": 80       
    }
}
"""

CONFIG = json.loads(STRATEGY_JSON)
TEST_MODE = True 

# ==========================================
# 2. 数据获取与指标计算 (强化文件加载逻辑)
# ==========================================
class MarketData:
    
    # --- 辅助函数：统一的本地数据加载 (支持 json/pkl) ---
    @staticmethod
    def _load_local_data(filename):
        file_path = Path(filename)
        if not file_path.exists(): return None
        try:
            if filename.endswith('.json'):
                with open(file_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    if isinstance(data, list): return data
            elif filename.endswith('.pkl'):
                data = pd.read_pickle(file_path)
                if isinstance(data, pd.DataFrame): return data
        except Exception as e:
            print(f"[严重警告] ❌ 本地文件 {filename} 格式错误或损坏: {e}")
            return None
        return None

    @staticmethod
    def _save_local_data(filename, data):
        try:
            if filename.endswith('.json'):
                with open(filename, 'w', encoding='utf-8') as f:
                    json.dump(data, f)
            elif filename.endswith('.pkl'):
                data.to_pickle(filename)
            print(f"[系统] 💾 数据已保存到本地: {filename}")
        except Exception as e:
            print(f"[警告] 缓存保存失败: {e}")

    @staticmethod
    def fetch_raw_data():
        p = CONFIG['params']
        filename = p['raw_cache_filename']
        limit = p['history_limit']
        use_cache = p.get('use_local_cache', False)
        
        # 1. 尝试加载本地原始 K 线缓存
        if use_cache:
            raw_data = MarketData._load_local_data(filename)
            if raw_data:
                if len(raw_data) >= limit:
                    print(f"[系统] ⚡ 已加载本地原始 K 线缓存: {filename} (共 {len(raw_data)} 条)")
                    return raw_data[-limit:]
                else:
                    print(f"[警告] 本地文件 {filename} 数据不足 ({len(raw_data)}/{limit})。尝试联网更新...")
            else:
                print(f"[警告] ⚠️ 未找到本地原始 K 线文件: {filename}。")

        # 2. 联网拉取 (此处应为完整的 API 拉取逻辑)
        print(f"[系统] 🌐 正在尝试从 API 拉取 {limit} 根数据...")
        all_data = []
        # *** 实际 API 请求代码块应放置在此处 ***
        # 由于无法执行网络请求，为避免程序中断，我们假设 API 请求返回空，并进行最终警告
        if not all_data:
            print(f"[严重警告] ❌ 联网拉取失败，且未找到有效的本地数据。无法进行回测。")
            return []
        # *** (API 请求代码结束) ***
        
        if len(all_data) >= limit:
            final_data = all_data[-limit:]
            if use_cache:
                MarketData._save_local_data(filename, final_data)
            return final_data
        
        return []

    # --- 核心修改：计算 20/60/120 EMA, MACD, RSI ---
    @staticmethod
    def calculate_and_cache_indicators(raw_data):
        p = CONFIG['params']
        processed_filename = p['processed_cache_filename']
        
        # 尝试加载已计算的指标表
        if p.get('use_local_cache', False):
            processed_df = MarketData._load_local_data(processed_filename)
            if isinstance(processed_df, pd.DataFrame):
                return processed_df

        # 数据处理
        df = pd.DataFrame(raw_data).iloc[:, :6]
        df.columns = ['time', 'open', 'high', 'low', 'close', 'volume']
        df = df.astype(float)
        
        print("[系统] ⚙️ 正在计算 20/60/120 EMA, MACD, RSI 指标...")

        # 1. 核心指标：均线 (20, 60, 120)
        df['ema_s'] = df['close'].ewm(span=p['ema_short'], adjust=False).mean()
        df['ema_m'] = df['close'].ewm(span=p['ema_mid'], adjust=False).mean()
        df['ema_l'] = df['close'].ewm(span=p['ema_long'], adjust=False).mean()

        # 2. 动能指标：MACD (用于止盈辅助)
        df['macd_fast'] = df['close'].ewm(span=p['macd_fast'], adjust=False).mean()
        df['macd_slow'] = df['close'].ewm(span=p['macd_slow'], adjust=False).mean()
        df['macd_line'] = df['macd_fast'] - df['macd_slow']
        df['signal_line'] = df['macd_line'].ewm(span=p['macd_signal'], adjust=False).mean()
        df['macd_hist'] = df['macd_line'] - df['signal_line']
        
        # 3. 超买超卖指标：RSI (用于止盈辅助)
        delta = df['close'].diff()
        gain = (delta.where(delta > 0, 0)).ewm(alpha=1/p['rsi_period'], adjust=False).mean()
        loss = (-delta.where(delta < 0, 0)).ewm(alpha=1/p['rsi_period'], adjust=False).mean()
        rs = gain / loss
        df['rsi'] = 100 - (100 / (1 + rs))

        MarketData._save_local_data(processed_filename, df)
        
        return df

# 3. 交易回测引擎 (均线突破和严格 SL/TP)
# ==========================================
class BacktestEngine:
    def __init__(self):
        self.balance = CONFIG['account']['initial_capital']
        self.position = 0 
        self.entry_price = 0
        self.leverage = CONFIG['account']['leverage']
        self.fee_rate = CONFIG['account']['fee_rate']
        self.logs = []
        self.markers = []
        self.initial_capital = CONFIG['account']['initial_capital']
        self.stop_loss_price = 0 

    def run(self, df):
        p = CONFIG['params']
        start_index = p['ema_long'] + 10 
        
        for i in range(start_index, len(df)):
            curr = df.iloc[i]
            prev = df.iloc[i-1]
            signal = "HOLD"
            
            # --- 0. 更新动态止损位 ---
            if self.position != 0:
                # 动态止损：多单止损设为 EMA60，空单止损设为 EMA60
                self.stop_loss_price = curr['ema_m'] 
                
                # --- 1. 风险检查 (止损 SL) ---
                if (self.position > 0 and curr['close'] < self.stop_loss_price):
                    self._execute_trade("CLOSE_SL", self.stop_loss_price, curr['time'])
                    continue
                elif (self.position < 0 and curr['close'] > self.stop_loss_price):
                    self._execute_trade("CLOSE_SL", self.stop_loss_price, curr['time'])
                    continue

                # --- 2. 动能止盈 (TP) ---
                # MACD 柱体开始收缩 (即当前柱体 < 前一柱体) 且远离0轴
                macd_decreasing = (abs(curr['macd_hist']) < abs(prev['macd_hist']))
                
                # RSI 达到超买/超卖区域
                rsi_exit_triggered = (
                    (self.position > 0 and curr['rsi'] >= p['rsi_profit_exit']) or
                    (self.position < 0 and curr['rsi'] <= (100 - p['rsi_profit_exit']))
                )
                
                if (macd_decreasing or rsi_exit_triggered):
                    self._execute_trade("CLOSE_TP", curr['close'], curr['time'])
                    continue
            
            # --- 3. 进场信号检查 (仅在无持仓时) ---
            if self.position == 0:
                # 计算均线密集度 (最大/最小 EMA 之间的价差占收盘价的百分比)
                emas = [curr['ema_s'], curr['ema_m'], curr['ema_l']]
                spread_pct = (max(emas) - min(emas)) / curr['close']
                is_coiled = spread_pct <= p['max_spread_pct']
                
                if is_coiled:
                    # 多头排列: 20 > 60 > 120
                    is_bullish_order = curr['ema_s'] > curr['ema_m'] > curr['ema_l']
                    # K 线站上 20 EMA
                    is_breakout_up = curr['close'] > curr['ema_s']
                    
                    # 空头排列: 20 < 60 < 120
                    is_bearish_order = curr['ema_s'] < curr['ema_m'] < curr['ema_l']
                    # K 线跌破 20 EMA
                    is_breakout_down = curr['close'] < curr['ema_s']

                    if is_breakout_up and is_bullish_order:
                        signal = "LONG"
                    elif is_breakout_down and is_bearish_order:
                        signal = "SHORT"
            
            # --- 4. 执行交易 ---
            if signal != "HOLD":
                self._execute_trade(signal, curr['close'], curr['time'])
        
        # 最终平仓
        if self.position != 0:
            last = df.iloc[-1]
            self._execute_trade("CLOSE", last['close'], last['time'])
            
        final_multiplier = self.balance / self.initial_capital
        print(f"[回测完成] 最终余额: {self.balance:.2f} U, 收益倍数: {final_multiplier:.2f}x")
        return self.logs, self.markers

    def _execute_trade(self, signal, price, timestamp):
        t_str = datetime.fromtimestamp(timestamp/1000).strftime('%Y-%m-%d %H:%M')
        fee_rate = self.fee_rate
        
        # --- 1. 平仓/结算逻辑 ---
        if self.position != 0 and signal in ["CLOSE", "CLOSE_SL", "CLOSE_TP"]:
            direction = "LONG" if self.position > 0 else "SHORT"
            qty = abs(self.position)
            value = qty * price
            fee = value * fee_rate
            
            pnl = (price - self.entry_price) * qty if direction == "LONG" else (self.entry_price - price) * qty
            self.balance += (pnl - fee)
            
            log_type = signal
            
            self.logs.append({"time": t_str, "type": log_type, "price": round(price, 2), "pnl": round(pnl, 2), "balance": round(self.balance, 2)})
            self.position = 0
            
            shape = "circle" if "SL" in log_type else ("square" if "TP" in log_type else "flag")
            color = "#f6465d" if "SL" in log_type else ("#0ecb81" if "TP" in log_type else "#f0b90b")
            self.markers.append({"time": timestamp / 1000, "position": "inBar", "color": color, "shape": shape, "text": f"{log_type}@{price:.0f}"})
            
        # --- 2. 开仓逻辑 ---
        if self.position == 0 and signal in ["LONG", "SHORT"]:
            margin = self.balance * 0.99
            notional = margin * self.leverage
            qty = notional / price
            fee = notional * fee_rate
            
            self.balance -= fee
            self.entry_price = price
            self.position = qty if signal == "LONG" else -qty
            
            self.logs.append({"time": t_str, "type": "OPEN " + signal, "price": round(price, 2), "pnl": 0, "balance": round(self.balance, 2)})
            
            color = "#0ecb81" if signal == "LONG" else "#f6465d"
            self.markers.append({"time": timestamp / 1000, "position": "belowBar" if signal == "LONG" else "aboveBar", "color": color, "shape": "arrowUp" if signal == "LONG" else "arrowDown", "text": f"{signal[0]}@{price:.0f}"})
